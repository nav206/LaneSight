<script>
  // Initialize Leaflet Map centered on SF Market Street
  const map = L.map('map', {
    zoomControl: false
  }).setView([37.7850, -122.4030], 14);

  // Clean vector base tile layer (CartoDB Positron)
  L.tileLayer('https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png', {
    maxZoom: 19,
    subdomains: 'abcd',
    attribution: '&copy; OpenStreetMap &copy; CARTO'
  }).addTo(map);

  // 1. ROUTE SEGMENTS (Color Coded Polylines)
  
  // Green Segment: Protected Two-Way Cycle Track (Market St)
  const safeSegment = [
    [37.7792, -122.4191], // Civic Center
    [37.7858, -122.4065]  // Towards 4th St
  ];
  L.polyline(safeSegment, { color: '#00c853', weight: 7, opacity: 0.9 }).addTo(map);

  // Yellow Segment: Cautious Shared Lane Segment
  const cautionSegment = [
    [37.7858, -122.4065],
    [37.7910, -122.3980]
  ];
  L.polyline(cautionSegment, { color: '#ffd600', weight: 7, opacity: 0.9 }).addTo(map);

  // Red Segment: High Hazard Segment near construction
  const hazardSegment = [
    [37.7910, -122.3980],
    [37.7935, -122.3930]  // Near Ferry Building
  ];
  L.polyline(hazardSegment, { color: '#ff3d00', weight: 7, opacity: 0.9 }).addTo(map);


  // 2. TAPPABLE ANNOTATION MARKERS
  
  // Custom Icon Helpers
  function createBadgeIcon(text, className) {
    return L.divIcon({
      className: className,
      html: text,
      iconSize: [32, 22],
      iconAnchor: [16, 11]
    });
  }

  // Annotation 1: Two-Way Cycle Track
  const ann1 = L.marker([37.7825, -122.4130], {
    icon: createBadgeIcon('⇄', 'leaflet-annotation-icon')
  }).addTo(map);
  
  ann1.bindPopup(`
    <b>⇄ Two-Way Protected Cycle Track</b><br/>
    <b>Surface:</b> Smooth Pavement<br/>
    <b>Barrier:</b> Raised Concrete Curb
  `);

  // Annotation 2: Center Running Lane
  const ann2 = L.marker([37.7885, -122.4020], {
    icon: createBadgeIcon('⬆️', 'leaflet-annotation-icon')
  }).addTo(map);
  
  ann2.bindPopup(`
    <b>⬆️ Center-Running Bike Lane</b><br/>
    <b>Notice:</b> Positioned between traffic lanes.<br/>
    Watch for turning vehicles at intersections.
  `);

  // Hazard Annotation: Bumpy Surface / Construction
  const hazardMarker = L.marker([37.7922, -122.3955], {
    icon: createBadgeIcon('⚠️', 'leaflet-hazard-icon')
  }).addTo(map);

  hazardMarker.bindPopup(`
    <b>⚠️ Surface Hazard: Rough Road</b><br/>
    Active road resurfacing. Slow down for loose gravel.
  `);


  // 3. REAL-TIME GBFS DOCK MARKERS (MOCK DOCK COUNTS)
  
  const dock1 = L.marker([37.7785, -122.4180], {
    icon: createBadgeIcon('📍 12', 'leaflet-dock-icon')
  }).addTo(map);
  dock1.bindPopup('<b>Civic Center Dock</b><br/>12 Open Lyft Bike Docks');

  const dock2 = L.marker([37.7860, -122.4050], {
    icon: createBadgeIcon('📍 8', 'leaflet-dock-icon')
  }).addTo(map);
  dock2.bindPopup('<b>Powell St Station Dock</b><br/>8 Open Lyft Bike Docks');

  const dock3 = L.marker([37.7940, -122.3920], {
    icon: createBadgeIcon('📍 15', 'leaflet-dock-icon')
  }).addTo(map);
  dock3.bindPopup('<b>Ferry Building Dock</b><br/>15 Open Lyft Bike Docks');

  // Trigger default preset on load
  function triggerSearch() {
    map.flyTo([37.7850, -122.4030], 14);
  }
</script>
